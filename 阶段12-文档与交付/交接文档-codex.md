```python
# 创建云机
POST /container_api/v1/create
{
    "user_name": "TK-001",
    "bool_start": true,
    "image_repository": "vcloud_android13_edge_20250925011125",
    "resolution": {"width": 1080, "height": 1920, "dpi": 420}

{"db_id": "EDGE001"}

POST /android_api/v1/proxy_set/{db_id}
{"proxyUrl": "socks5://user:pass@proxy.com:1080"}

# 批量安装APK
POST /android_api/v1/upload_file_android_batch
files={'file': open('tiktok.apk', 'rb')}
data={'db_ids': 'EDGE001,EDGE002'}
```

